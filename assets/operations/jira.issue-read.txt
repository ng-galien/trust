Operation "jira.issue-read"

Input
  issue

Environment
  jiraBaseUrl
  jiraAuthorization

ticket = HTTP
  GET jiraBaseUrl & "/rest/api/3/issue/" & issue
  Accept: "application/json"
  Authorization: jiraAuthorization

Fail if ticket.status < 200 or ticket.status >= 300

Produce with JSONata
(
  $type := $lowercase(ticket.body.fields.issuetype.name);
  $status := $replace($lowercase(ticket.body.fields.status.name), /[_ ]+/, "-");
  {
    "summary": ticket.body.fields.summary,
    "description": $join(ticket.body.fields.description.**.text, ""),
    "issueType": $type = "bug" ? "defect" : $type,
    "workflowStatus": $status = "to-do" ? "todo" : $status
  }
)
