Operation "maven.project-verify"

Input
  revision

Environment
  projectRoot

beforeHead = Shell
  git rev-parse --verify HEAD
  cwd: projectRoot

beforeTree = Shell
  git status --porcelain=v1 --untracked-files=normal
  cwd: projectRoot

Fail if beforeHead.exitCode != 0 or $trim(beforeHead.stdout) != revision
Fail if beforeTree.exitCode != 0 or $trim(beforeTree.stdout) != ""

verify = Shell
  mvn -B -ntp clean verify
  cwd: projectRoot
  allow non-zero exit

afterHead = Shell
  git rev-parse --verify HEAD
  cwd: projectRoot

afterTree = Shell
  git status --porcelain=v1 --untracked-files=normal
  cwd: projectRoot

Fail if afterHead.exitCode != 0 or $trim(afterHead.stdout) != revision
Fail if afterTree.exitCode != 0 or $trim(afterTree.stdout) != ""

Produce with JSONata
{
  "verifiedRevision": revision,
  "verificationStatus": verify.exitCode = 0 ? "successful" : "failed"
}
